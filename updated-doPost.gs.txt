function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var values = sheet.getDataRange().getValues();
  if (values.length < 1) {
    return ContentService.createTextOutput(JSON.stringify([]))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var headers = values[0];
  var records = [];
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    var record = {};
    for (var j = 0; j < headers.length; j++) {
      record[headers[j]] = row[j];
    }
    records.push(record);
  }
  return ContentService.createTextOutput(JSON.stringify(records))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);

  // If the sheet is completely empty, create the header row from this record.
  if (sheet.getLastRow() === 0) {
    var headers = Object.keys(data).filter(function(k){ return k !== 'action'; });
    sheet.appendRow(headers);
  }

  var headerRow = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];

  // --- NEW: auto-add any columns that exist in the incoming data
  // but are missing from the current header row. This means new
  // form fields (added later) will automatically get their own
  // column instead of being silently dropped. ---
  var incomingKeys = Object.keys(data).filter(function(k){ return k !== 'action'; });
  var missingKeys = incomingKeys.filter(function(k){ return headerRow.indexOf(k) === -1; });
  if (missingKeys.length > 0) {
    var startCol = headerRow.length + 1;
    sheet.getRange(1, startCol, 1, missingKeys.length).setValues([missingKeys]);
    headerRow = headerRow.concat(missingKeys);
  }

  if (data.action === 'update' && data.id) {
    var values = sheet.getDataRange().getValues();
    var idCol = headerRow.indexOf('id');
    for (var i = 1; i < values.length; i++) {
      if (values[i][idCol] == data.id) {
        var newRow = headerRow.map(function(h){
          return data[h] !== undefined ? data[h] : (values[i][headerRow.indexOf(h)] || '');
        });
        sheet.getRange(i + 1, 1, 1, headerRow.length).setValues([newRow]);
        break;
      }
    }
  } else {
    var newRow = headerRow.map(function(h){ return data[h] !== undefined ? data[h] : ''; });
    sheet.appendRow(newRow);
  }

  return ContentService.createTextOutput(JSON.stringify({result:"success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
